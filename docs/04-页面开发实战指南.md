# SD-WAN 管理平台 — 页面开发实战指南

> **这份文档手把手教你做每一个页面。**
> 按照文档顺序，从简单到复杂，一个一个做。

---

## 开发顺序

```
第 1 个：登录页         （最简单，只有表单）
第 2 个：注册页         （跟登录页类似）
第 3 个：首页/总览       （展示数据，没有交互）
第 4 个：站点管理       （完整 CRUD，最核心）
第 5 个：设备管理       （复用站点管理模式）
第 6 个：链路管理       （表格页）
第 7 个：告警中心       （表格页 + 状态操作）
第 8 个：策略管理       （表单页）
第 9 个：拓扑图         （可视化，需要学 g6）
第 10 个：全网监控大屏   （图表，需要学 Recharts）
第 11 个：系统设置      （表单页）
```

```tsx
// src/pages/Login/index.tsx
import { useState } from 'react';
import { Form, Input, Button, Card, Typography, Checkbox, Alert } from 'antd';
import { UserOutlined, LockOutlined } from '@ant-design/icons';
import { useNavigate, Link } from 'react-router-dom';
import './Login.css';

const { Title, Text } = Typography;

export default function Login() {
    const navigate = useNavigate();
    const [isLoading, setIsLoading] = useState(false);
    const [error, setError] = useState('');

    const handleSubmit = async (values) => {
        setIsLoading(true);
        setError('');
        try {
            // TODO: 调用登录 API
            await new Promise(r => setTimeout(r, 1000)); // 模拟延迟
            navigate('/'); // 登录成功跳转首页
        } catch (e) {
            setError(e.message || '登录失败');
        } finally {
            setIsLoading(false);
        }
    };

    return (
        <div className="login-page">
            <Card className="login-card">
                {/* 品牌标识 */}
                <div className="login-brand">
                    <div className="login-brand-logo">SD</div>
                    <Title level={2} className="login-brand-title">SD-WAN</Title>
                </div>

                {/* 错误提示 */}
                {error && (
                    <Alert message={error} type="error" showIcon closable
                        onClose={() => setError('')} style={{ marginBottom: 16 }} />
                )}

                {/* 表单 */}
                <Form layout="vertical" size="large" onFinish={handleSubmit}>
                    <Form.Item name="username"
                        rules={[{ required: true, message: '请输入用户名' }, { min: 3, message: '至少3个字符' }]}>
                        <Input prefix={<UserOutlined />} placeholder="用户名" />
                    </Form.Item>
                    <Form.Item name="password"
                        rules={[{ required: true, message: '请输入密码' }, { min: 6, message: '至少6个字符' }]}>
                        <Input.Password prefix={<LockOutlined />} placeholder="密码" />
                    </Form.Item>
                    <Form.Item>
                        <div style={{ display: 'flex', justifyContent: 'space-between' }}>
                            <Form.Item name="remember" valuePropName="checked" noStyle>
                                <Checkbox>记住用户名</Checkbox>
                            </Form.Item>
                        </div>
                    </Form.Item>
                    <Form.Item>
                        <Button type="primary" htmlType="submit" loading={isLoading} block>登录</Button>
                    </Form.Item>
                </Form>

                <div className="login-footer">
                    <Text type="secondary">还没有账号？</Text>
                    <Link to="/register">去注册</Link>
                </div>
            </Card>
        </div>
    );
}
```

### 核心代码结构

```tsx
export default function SiteManagement() {
    // 状态
    const [data, setData] = useState(MOCK_DATA);
    const [search, setSearch] = useState('');
    const [regionFilter, setRegionFilter] = useState('');
    const [statusFilter, setStatusFilter] = useState('');
    const [modalOpen, setModalOpen] = useState(false);
    const [editing, setEditing] = useState(null);

    // 统计
    const stats = {
        total: data.length,
        online: data.filter(s => s.status === 'online').length,
        offline: data.filter(s => s.status === 'offline').length,
        warning: data.filter(s => s.status === 'warning').length,
    };

    // 筛选
    const filtered = data.filter(s => {
        if (search && !s.displayName.includes(search)) return false;
        if (regionFilter && s.region !== regionFilter) return false;
        if (statusFilter && s.status !== statusFilter) return false;
        return true;
    });

    // 删除
    const handleDelete = (id) => {
        Modal.confirm({
            title: '确认删除',
            content: '删除后不可恢复',
            onOk: () => {
                setData(prev => prev.filter(s => s.id !== id));
                message.success('删除成功');
            },
        });
    };

    // 保存（新增/编辑）
    const handleSave = (values) => {
        if (editing) {
            setData(prev => prev.map(s => s.id === editing.id ? { ...s, ...values } : s));
            message.success('编辑成功');
        } else {
            setData(prev => [...prev, { id: Date.now(), ...values }]);
            message.success('新增成功');
        }
        setModalOpen(false);
        setEditing(null);
    };

    // 表格列
    const columns = [
        { title: '站点名称', dataIndex: 'displayName', render: (text, record) => (
            <div>
                <div style={{ fontWeight: 600 }}>{text}</div>
                <div style={{ fontSize: 11, color: 'var(--text-muted)' }}>{record.name}</div>
            </div>
        )},
        { title: '区域', dataIndex: 'region', render: v => REGION_MAP[v] },
        { title: '状态', dataIndex: 'status', render: v => <Tag color={STATUS_MAP[v].color}>{STATUS_MAP[v].label}</Tag> },
        { title: '延迟', dataIndex: 'latency', render: v => `${v}ms` },
        { title: '操作', render: (_, r) => (
            <Space>
                <Button size="small" onClick={() => { setEditing(r); setModalOpen(true); }}>编辑</Button>
                <Button size="small" danger onClick={() => handleDelete(r.id)}>删除</Button>
            </Space>
        )},
    ];

    return (
        <div className="site-page">
            {/* 标题 */}
            <div className="page-header">
                <div><h1>站点管理</h1><span>管理所有站点</span></div>
                <Button type="primary" onClick={() => { setEditing(null); setModalOpen(true); }}>新增站点</Button>
            </div>

            {/* 统计卡片 */}
            <div className="overview-cards">
                <div className="overview-card"><div className="overview-card__value">{stats.total}</div><div className="overview-card__label">总站点</div></div>
                {/* 更多卡片... */}
            </div>

            {/* 筛选栏 */}
            <div className="filter-bar">
                <div className="filter-bar__left">
                    <Input placeholder="搜索" prefix={<SearchOutlined />} value={search} onChange={e => setSearch(e.target.value)} />
                    <Select placeholder="区域" onChange={setRegionFilter} options={[...]} />
                    <Select placeholder="状态" onChange={setStatusFilter} options={[...]} />
                </div>
                <span>共 {filtered.length} 个站点</span>
            </div>

            {/* 表格 */}
            <div className="table-wrapper">
                <Table dataSource={filtered} columns={columns} rowKey="id" pagination={{ pageSize: 10 }} />
            </div>

            {/* 弹窗 */}
            <SiteFormModal open={modalOpen} editingSite={editing} onCancel={() => setModalOpen(false)} onSave={handleSave} />
        </div>
    );
}
```

### 弹窗组件

```tsx
// SiteFormModal.tsx
export default function SiteFormModal({ open, editingSite, onCancel, onSave }) {
    const [form] = Form.useForm();

    useEffect(() => {
        if (open && editingSite) {
            form.setFieldsValue(editingSite); // 编辑时填充数据
        } else {
            form.resetFields(); // 新增时清空
        }
    }, [open, editingSite]);

    const handleOk = async () => {
        const values = await form.validateFields();
        onSave(values);
    };

    return (
        <Modal title={editingSite ? '编辑站点' : '新增站点'} open={open} onOk={handleOk} onCancel={onCancel}>
            <Form form={form} layout="vertical">
                <Form.Item name="displayName" label="站点名称" rules={[{ required: true }]}>
                    <Input placeholder="如：北京总部" />
                </Form.Item>
                <Form.Item name="name" label="站点编码" rules={[{ required: true }]}>
                    <Input placeholder="如：site-bj-01" />
                </Form.Item>
                <Form.Item name="region" label="区域" rules={[{ required: true }]}>
                    <Select options={[{ label: '中国', value: 'CN' }, { label: '新加坡', value: 'SG' }]} />
                </Form.Item>
            </Form>
        </Modal>
    );
}
```

```tsx
import { useEffect, useRef } from 'react';
import { Graph } from '@antv/g6';

export default function TopologyView() {
    const containerRef = useRef(null);

    useEffect(() => {
        const graph = new Graph({
            container: containerRef.current,
            width: containerRef.current.offsetWidth,
            height: containerRef.current.offsetHeight,
            modes: { default: ['drag-canvas', 'zoom-canvas', 'drag-node'] },
            defaultNode: {
                type: 'circle',
                size: 40,
                style: { fill: '#00b4d8', stroke: '#00b4d8' },
            },
            defaultEdge: {
                type: 'line',
                style: { stroke: '#1e2a5c', lineWidth: 2 },
            },
        });

        // 数据
        const data = {
            nodes: [
                { id: 'hq', label: '北京总部', x: 400, y: 100, size: 60 },
                { id: 'sh', label: '上海分支', x: 200, y: 300, size: 40 },
                { id: 'gz', label: '广州分支', x: 600, y: 300, size: 40 },
            ],
            edges: [
                { source: 'hq', target: 'sh' },
                { source: 'hq', target: 'gz' },
            ],
        };

        graph.data(data);
        graph.render();

        return () => graph.destroy(); // 清理
    }, []);

    return <div ref={containerRef} style={{ width: '100%', height: '100%' }} />;
}
```
